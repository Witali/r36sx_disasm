#!/bin/sh

mount -t tmpfs tmpfs /media

if ! grep -q devtmpfs /proc/mounts; then
        if grep -q devtmpfs /proc/filesystems; then
                mount -t devtmpfs devtmpfs /dev
        else
                mount -t tmpfs tmpfs /dev  -o size=64k,mode=0755
                rm /dev/null
                mknod /dev/null c 1 3
                mkdir /dev/pts /dev/shm
                mount -t devpts devpts /dev/pts
        fi
fi

echo > /dev/mdev.seq

echo "/sbin/mdev" >/proc/sys/kernel/hotplug

touch /dev/.udev

[ -d "/sys/kernel" ] || mount sysfs /sys -t sysfs

mdev -s

=== etc/init.d/S99app ===
#!/bin/sh
#
# Start the preinit
#

MNTDIR=""
wait_for_media_ready()
{
	while true
	do
		for subdir in $(ls /media); do
			if [ -e /media/$subdir/cubegm/icube ] ; then
				MNTDIR=$subdir
				return
			fi
		done

		sleep 0.5
	done
}

wait_for_media_ready

#mount
echo "mount --bind /media/${MNTDIR} /mnt/sdcard"
mount --bind /media/${MNTDIR} /mnt/sdcard
memsize=$(free | grep Mem | awk '{print $2}')

if [ -d /media/${MNTDIR}/rootfs ] ; then
	echo "mount --bind /media/${MNTDIR}/rootfs/lib /lib"
	mount --bind /media/${MNTDIR}/rootfs/lib /lib
	echo "mount --bind /media/${MNTDIR}/rootfs/usr /usr"
	mount --bind /media/${MNTDIR}/rootfs/usr /usr
	echo "mount --bind /media/${MNTDIR}/rootfs/bin /bin"
	mount --bind /media/${MNTDIR}/rootfs/bin /bin
	echo "mount --bind /media/${MNTDIR}/rootfs/sbin /sbin"
	mount --bind /media/${MNTDIR}/rootfs/sbin /sbin
	echo "mount --bind /media/${MNTDIR}/rootfs/etc /etc"
	mount --bind /media/${MNTDIR}/rootfs/etc /etc
	echo "mount --bind /media/${MNTDIR}/rootfs/hcdemo_files /hcdemo_files"
	mount --bind /media/${MNTDIR}/rootfs/hcdemo_files /hcdemo_files
else
	mount --bind /media/${MNTDIR}/cubegm/lib /lib
	echo "mount --bind /media/${MNTDIR}/cubegm/lib /lib"
	mount --bind /media/${MNTDIR}/cubegm/usr/lib /usr/lib
	echo "mount --bind /media/${MNTDIR}/cubegm/usr/lib /usr/lib"
fi

if [ $memsize -le 128000 ] ; then
	if [ ! -f /mnt/sdcard/cubegm/pagefile.sys ] ; then
		echo "Not found pagefile.sys, rebooting ..."
		echo "Creating pagefile.sys ..."
		fallocate -l 128M /mnt/sdcard/cubegm/pagefile.sys
		mkswap /mnt/sdcard/cubegm/pagefile.sys
		echo "Creating pagefile.sys done"
	fi
	if [ -f /mnt/sdcard/cubegm/pagefile.sys ] ; then
		echo "Swapon ..."
		swapon /mnt/sdcard/cubegm/pagefile.sys
		if [ $? -eq 0 ] && [ $(free | grep Swap | awk '{print $2}') -gt 0 ]; then
			echo "Swap space activated successfully"
			/mnt/sdcard/cubegm/icube.sh &
		else
			echo "Failed to activate swap space"
			# You can add error handling here if needed
		fi
	fi
else
/mnt/sdcard/cubegm/icube.sh &
fi


=== etc/init.d/rcK ===
#!/bin/sh


# Stop all init scripts in /etc/init.d
# executing them in reversed numerical order.
#
for i in $(ls -r /etc/init.d/S??*) ;do

     # Ignore dangling symlinks (if any).
     [ ! -f "$i" ] && continue

     case "$i" in
	*.sh)
	    # Source shell script for speed.
	    (
		trap - INT QUIT TSTP
		set stop
		. $i
	    )
	    ;;
	*)
	    # No sh extension, so fork subprocess.
	    $i stop
	    ;;
    esac
done

# Copyright (C) 2001 Erik Andersen <andersen@codepoet.org>
#
# Note: BusyBox init doesn't support runlevels.  The runlevels field is
# completely ignored by BusyBox init. If you want runlevels, use
# sysvinit.
#
# Format for each entry: <id>:<runlevels>:<action>:<process>
#
# id        == tty to run on, or empty for /dev/console
# runlevels == ignored
# action    == one of sysinit, respawn, askfirst, wait, and once
# process   == program to run

# Startup the system
::sysinit:/bin/mount -t proc proc /proc
::sysinit:/bin/mount -o remount,rw /
::sysinit:/bin/mkdir -p /dev/pts /dev/shm
::sysinit:/bin/mount -a
::sysinit:/sbin/swapon -a
null::sysinit:/bin/ln -sf /proc/self/fd /dev/fd
null::sysinit:/bin/ln -sf /proc/self/fd/0 /dev/stdin
null::sysinit:/bin/ln -sf /proc/self/fd/1 /dev/stdout
null::sysinit:/bin/ln -sf /proc/self/fd/2 /dev/stderr
::sysinit:/bin/hostname -F /etc/hostname
# now run any rc scripts

=== init ===
#!/bin/sh
# devtmpfs does not get automounted for initramfs
/bin/mount -t devtmpfs devtmpfs /dev

# use the /dev/console device node from devtmpfs if possible to not
# confuse glibc's ttyname_r().
# This may fail (E.G. booted with console=), and errors from exec will
# terminate the shell, so use a subshell for the test
if (exec 0</dev/console) 2>/dev/null; then
    exec 0</dev/console
    exec 1>/dev/console
    exec 2>/dev/console
fi

exec /sbin/init "$@"
